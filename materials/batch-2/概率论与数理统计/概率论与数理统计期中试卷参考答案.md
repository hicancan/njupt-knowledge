   \end{array} 
   \right., \quad
   f_{Y}(y) = \left\{ 
   \begin{array}{ll}
   e^{-y} & y > 0\\ 
   0 & \text{其它} 
   \end{array} 
   \right.
   $$
2. 数字特征：
   $EX = \frac{1}{2}, EY = 1, E(XY) = \frac{1}{2};$
   $Cov(X,Y) = 0, DX = \frac{1}{4}, DY = 1;$

   $$
   \begin{array}{c c c} 
   U \backslash V & 0 & 1 \\ 
   \hline 
   0 & \frac {1}{4} & \frac {1}{4} \\ 
   1 & 0 & \frac {1}{2} 
   \end{array}
   $$
