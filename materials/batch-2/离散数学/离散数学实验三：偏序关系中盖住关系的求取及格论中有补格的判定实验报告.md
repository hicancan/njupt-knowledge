## 二、实验目的和要求

### 实验目的

对任意给定正整数，利用整除关系求所有由其因子构成的集合所构成的格，判断其是否为有补格。

## 四、实验原理及内容

```cpp
// ConsoleApplication1.cpp：定义控制台应用程序的入口点。
#include "stdafx.h"
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

// 获取 n 的所有因子
vector<int> getDivisors(int n) {
    vector<int> divisors;
    for (int i = 1; i <= n; ++i) {
        if (n % i == 0) {
            divisors.push_back(i);
        }
    }
    return divisors;
}

// 判断 a 是否盖住 b (在整除关系下)
// 即 b | a, 且不存在 c 使得 b | c 且 c | a (b != c, c != a)
bool covers(const vector<int>& divisors, int a, int b) {
    if (a % b != 0) return false;
    if (a == b) return false;
    
    // 检查是否存在中间元素
    for (int c : divisors) {
        if (c == a || c == b) continue;
        if (a % c == 0 && c % b == 0) {
            return false; // 找到了中间元素，所以 a 不盖住 b
        }
    }
    return true;
}

// 判断格是否有补元
// 对于格中的每个元素 x，如果存在 y 使得 x ^ y = 0 (最小元) 且 x v y = 1 (最大元)，则 x 有补元
// 这里假设最小元是 1，最大元是 n
bool hasComplement(const vector<int>& divisors, int n) {
    int minElement = 1;
    int maxElement = n;
    
    for (int x : divisors) {
        bool foundComplement = false;
        for (int y : divisors) {
            // 计算最大公约数作为交运算 (x ^ y)
            // 计算最小公倍数作为并运算 (x v y)
            int gcd_xy = __gcd(x, y);
            int lcm_xy = (x * y) / __gcd(x, y);
            
            if (gcd_xy == minElement && lcm_xy == maxElement) {
                foundComplement = true;
                break;
            }
        }
        if (!foundComplement) {
            return false; // 存在元素没有补元
        }
    }
    return true;
}

int main() {
    int n;
    cout << "请输入一个正整数: ";
    cin >> n;
    
    vector<int> divisors = getDivisors(n);
    
    cout << "因子集合: {";
    for (size_t i = 0; i < divisors.size(); ++i) {
        cout << divisors[i];
        if (i < divisors.size() - 1) cout << ", ";
    }
    cout << "}" << endl;
    
    cout << "盖住关系如下:" << endl;
    for (int a : divisors) {
        for (int b : divisors) {
            if (covers(divisors, a, b)) {
                cout << a << " 盖住 " << b << endl;
            }
        }
    }
    
    bool isComplemented = hasComplement(divisors, n);
    cout << "该格是有补格吗? " << (isComplemented ? "是" : "否") << endl;
    
    return 0;
}
```
